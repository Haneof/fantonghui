name: CI - M0-001 boundaries

on:
  push:
    branches: [ "aios-2.0", "arena/*" ]
  pull_request:
    branches: [ "aios-2.0" ]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Install
        run: |
          python -m pip install --upgrade pip
          pip install -e ".[dev]"
      - name: Run pytest
        run: |
          python -m pytest -v
