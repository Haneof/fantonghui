## 附：FTS5 中文分词隔离复现（内存表，最小用例）

import sqlite3
c=sqlite3.connect(":memory:")
for tok in ["trigram","trigram case_sensitive 1"]:
    c.execute(f"CREATE VIRTUAL TABLE t USING fts5(x, tokenize=\"{tok}\")")
    c.execute("INSERT INTO t VALUES('第0条记录：今天和妈妈聊到生日，顺便说起礼物的事情')")
    c.execute("INSERT INTO t VALUES('加班熬夜导致心悸，妈妈送的礼物是按摩仪')")
    for q in ["妈妈","礼物","妈妈 礼物","和妈妈聊","按摩仪","生日礼物"]:
        print(tok, repr(q), c.execute("SELECT count(*) FROM t WHERE t MATCH ?",(q,)).fetchone()[0])
